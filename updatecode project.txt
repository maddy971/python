def update_payment(request):
    if request.method=="POST":
        id=request.POST.get('id')
        Request_id = request.POST.get('t1', '')
        Date = request.POST.get('t2', '')
        No_of_seats = request.POST.get('t3', '')
        costperseat = request.POST.get('t4', '')
        total = request.POST.get('t5', '')
        status = request.POST.get('t6', '')
        Payment.objects.filter(id=id).update(Request_id=Request_id, Date=Date, No_of_seats=No_of_seats, Cost_per_seat=costperseat,
                       Total=total, status=status)
        userdict = Payment.objects.all()
        return render(request, 'viewpayment.html', {'userdict': userdict})
